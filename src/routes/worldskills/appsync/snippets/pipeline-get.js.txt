import { util } from '@aws-appsync/utils';
import * as ddb from '@aws-appsync/utils/dynamodb';

export function request(ctx) {
  return ddb.get({ key: ctx.prev.result, consistentRead: true });
}

export function response(ctx) {
  if (ctx.error) util.error(ctx.error.message, ctx.error.type);
  if (!ctx.result) util.error('Item not found', 'NotFound');
  if (ctx.result.owner !== ctx.stash.owner) util.unauthorized();
  ctx.stash.item = ctx.result;
  return { id: ctx.result.id, title: ctx.result.title };
}
